  <script>
    let transactions = [
      { id: 1, title: 'Malzeme Satışı', category: 'Satış', amount: 15400, type: 'income', date: 'Bugün' },
      { id: 2, title: 'Kira Ödemesi', category: 'Kira', amount: -8500, type: 'expense', date: 'Dün' },
      { id: 3, title: 'Kablo Tedariği', category: 'Tedarikçi', amount: -12300, type: 'expense', date: '03 Ekm' }
    ];

    function renderTransactions() {
      const list = document.getElementById('transaction-list');
      list.innerHTML = transactions.map(t => `
        <div class="flex items-center justify-between py-2 border-b border-gray-50 last:border-none">
          <div class="flex items-center gap-3">
            <div class="w-9 h-9 rounded-full ${t.type === 'income' ? 'bg-emerald-100 text-emerald-600' : 'bg-rose-100 text-rose-600'} flex items-center justify-center">
              <i data-lucide="${t.type === 'income' ? 'arrow-down-left' : 'arrow-up-right'}" class="w-5 h-5"></i>
            </div>
            <div>
              <p class="font-semibold text-sm text-gray-800">${t.title}</p>
              <p class="text-xs text-gray-400">${t.category} • ${t.date}</p>
            </div>
          </div>
          <p class="font-bold text-sm ${t.type === 'income' ? 'text-emerald-600' : 'text-gray-900'}">
            ${t.type === 'income' ? '+' : ''}₺${Math.abs(t.amount).toLocaleString('tr-TR')}
          </p>
        </div>
      `).join('');
      lucide.createIcons();
    }

    function addTransactionPrompt() {
      const title = prompt("İşlem Başlığı:");
      if (!title) return;
      const amount = parseFloat(prompt("Tutar (₺):"));
      if (isNaN(amount)) return;
      const type = amount >= 0 ? 'income' : 'expense';

      transactions.unshift({
        id: Date.now(),
        title: title,
        category: 'Genel',
        amount: amount,
        type: type,
        date: 'Şimdi'
      });
      renderTransactions();
    }

    renderTransactions();
  </script>
